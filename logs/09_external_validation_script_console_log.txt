# ==============================================================================
> # Script: 12_continuum_panel.R
> # Purpose: Select a panel that classifies the oral carcinogenesis continuum as
> #          four states rather than two, and compare that selection against the
> #          binary selection the field publishes.
> #
> # THE THREE TRAINED STATES
> #
> #     donor    normal mucosa from a person without oral cancer, 54 arrays
> #     field    normal-appearing mucosa from a person who has oral cancer,
> #              pooling the contralateral, adjacent, and matched labels, 63 arrays
> #     cancer   carcinoma, 329 arrays
> #
> #   Dysplasia is excluded from training. Seventeen arrays cannot support a
> #   fitted class, and a class that small would set the penalty for every other
> #   class in a multinomial fit. Those arrays are scored afterwards in step 13,
> #   where a state the model never saw provides an independent check on the
> #   ordering rather than a fourth fitted parameter.
> #
> # WHY THE FIELD STATE IS THE POINT
> #   A pathologist reports donor mucosa and field mucosa identically. Both are
> #   normal squamous epithelium under a microscope. The distinction exists in the
> #   deposit's provenance record and nowhere in the histology. Every published
> #   pooled analysis assigns both to one normal class.
> #
> #   A colorectal continuum classifier separates normal, inflammatory bowel
> #   disease, adenoma, and carcinoma, and a pathologist can already stage those
> #   four. The oral trajectory carries a state that histology cannot see. That
> #   state is what this panel resolves, and it is the state with enough arrays to
> #   fit.
> #
> # CLASSIFIER CONFIGURATIONS
> #   Stability selection runs in several parallel configurations, following the
> #   colorectal design. The multinomial configuration asks which genes separate
> #   all four states at once. The one-versus-rest configuration for the field
> #   state asks which genes mark it specifically. The three adjacent pairwise
> #   configurations ask which genes mark each single transition along the order.
> #   Genes reaching the selection threshold in any configuration form the union
> #   set, which describes the trajectory. The multinomial subset forms the panel
> #   that classifies it.
> #
> # COUNTS
> #   The three trained states hold 54, 63, and 329 arrays. Every split and every
> #   bootstrap is stratified, so the class proportions are preserved throughout.
> #
> # INPUT   arm matrices, the manifest
> # OUTPUT  results/continuum_panels.rds, results/continuum_frequencies.rds,
> #         results/continuum_<config>_<arm>.csv, results/continuum_summary.csv
> # ==============================================================================

> continuum_label <- function(m) {
+   lab <- rep(NA_character_, nrow(m))
+   lab[m$class == "cancer"] <- "cancer"
+   lab[m$class == "normal" & m$normal_provenance == "donor"] <- "donor"
+   lab[m$class == "normal" & m$normal_provenance != "donor"] <- "field"
+   factor(lab, levels = CONTINUUM)

> # ------------------------------------------------------------------------------
> # Stability selection, binomial or multinomial
> # ------------------------------------------------------------------------------
> run_stability <- function(X, y, family, n_boot = N_BOOT) {
+   genes <- colnames(X)
+ 
+   boot_one <- function(b) {
+     idx <- unlist(lapply(levels(y), function(lev) {
+       i <- which(y == lev); if (!length(i)) return(integer(0))
+       sample(i, length(i), replace = TRUE)
+     }))
+     Xb <- X[idx, , drop = FALSE]; yb <- droplevels(y[idx])
+     if (nlevels(yb) < 2) return(NULL)
+ 
+     fold <- rep(NA_integer_, length(yb))
+     for (lev in levels(yb)) {
+       li <- which(yb == lev)
+       fold[li] <- sample(rep(seq_len(N_FOLDS), length.out = length(li)))
+     }
+     fit <- tryCatch(cv.glmnet(Xb, yb, family = family, alpha = ALPHA,
+                               type.measure = "class", foldid = fold),
+                     error = function(e) NULL)
+     if (is.null(fit)) return(NULL)
+     co <- coef(fit, s = "lambda.1se")
+     sel <- if (is.list(co))
+       unique(unlist(lapply(co, function(m) rownames(m)[which(as.numeric(m) != 0)])))
+     else rownames(co)[which(co[, 1] != 0)]
+     setdiff(sel, "(Intercept)")
+   }
+ 
+   res <- if (N_WORKERS <= 1L) {
+     lapply(seq_len(n_boot), boot_one)
+   } else if (.Platform$OS.type == "windows") {
+     cl <- parallel::makeCluster(N_WORKERS)
+     on.exit(parallel::stopCluster(cl), add = TRUE)
+     parallel::clusterEvalQ(cl, suppressPackageStartupMessages(library(glmnet)))
+     parallel::clusterExport(cl, c("X", "y", "family"), envir = environment())
+     parallel::clusterExport(cl, c("ALPHA", "N_FOLDS"), envir = globalenv())
+     parallel::clusterSetRNGStream(cl, 123)
+     parallel::parLapply(cl, seq_len(n_boot), boot_one)
+   } else {
+     mclapply(seq_len(n_boot), boot_one, mc.cores = N_WORKERS, mc.set.seed = TRUE)
+   }
+ 
+   counts <- setNames(rep(0, length(genes)), genes)
+   eff <- sum(!vapply(res, is.null, logical(1)))
+   for (r in res) if (!is.null(r)) counts[r] <- counts[r] + 1
+   list(freq = counts / eff, effective = eff)
+ }

> # Subset and outcome for one configuration
> config_data <- function(cfg, samples, lab) {
+   if (cfg == "multinomial")
+     return(list(keep = rep(TRUE, length(samples)), y = lab, family = "multinomial"))
+   if (grepl("^ovr_", cfg)) {
+     target <- sub("^ovr_", "", cfg)
+     y <- factor(ifelse(lab == target, target, "rest"), levels = c("rest", target))
+     return(list(keep = rep(TRUE, length(samples)), y = y, family = "binomial"))
+   }
+   parts <- strsplit(sub("^pair_", "", cfg), "_")[[1]]
+   keep <- lab %in% parts
+   list(keep = keep, y = factor(as.character(lab[keep]), levels = parts),
+        family = "binomial")
+ }
>
